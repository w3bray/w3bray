// safe_example.c
// Demonstrates loading a local library in the current process.
// This is a harmless example for learning the API shape.
#include <windows.h>
#include <stdio.h>

int main(void) {
    HMODULE hMod = LoadLibraryA("user32.dll");
    if (!hMod) {
        printf("LoadLibraryA failed: %lu\n", GetLastError());
        return 1;
    }

    FARPROC proc = GetProcAddress(hMod, "MessageBoxA");
    if (!proc) {
        printf("GetProcAddress failed: %lu\n", GetLastError());
        FreeLibrary(hMod);
        return 1;
    }

    printf("Library loaded successfully. Function pointer acquired.\n");
    FreeLibrary(hMod);
    return 0;
}
